/* Icons are painted by tools/build_tileset.py; the towel bundle by tools/build_model.py. The other models are vanilla
   stand-ins. */
module TraditionalPreservationDairy
{
    imports
    {
        Base,
    }

    /* A pot, jar or bucket wrapped in towels lies on the ground as one bundle, sized to the container it holds. The
       sizes come from vanilla's meshes: each model's bounding box times its scale over its file's unit, with a
       vanilla bucket taken as 30 cm tall; the bundle stands 1.15 times the container's height below the tie (0.7
       of the bundle) or 1.1 times its width, whichever is bigger, at 0.54 a metre (the series' glb models). The
       clay jar is tall and narrow: it has the narrow mesh (0.4 as wide, TowelBundleTall), at the height and width the
       user picked in game (2026-11-13; by the rule above the bundle came out twice as wide as the jar).
       Pot 45 x 35 cm, forged pot 46 x 36, glass jar 12 x 16, crafted jar 13 x 17, clay jar 31 x 49, bucket 29 x 30,
       carved bucket 35 x 38. Lua sets them on the item's modData worldStaticModel (M.WRAPPED_LOOKS). */
    model TowelBundle_Pot
    {
        mesh = TowelBundle,
        texture = TowelBundleModel,
        scale = 0.309,
    }

    model TowelBundle_PotForged
    {
        mesh = TowelBundle,
        texture = TowelBundleModel,
        scale = 0.318,
    }

    model TowelBundle_Jar
    {
        mesh = TowelBundle,
        texture = TowelBundleModel,
        scale = 0.145,
    }

    model TowelBundle_ClayJar
    {
        mesh = TowelBundleTall,
        texture = TowelBundleModel,
        scale = 0.42,
    }

    model TowelBundle_Bucket
    {
        mesh = TowelBundle,
        texture = TowelBundleModel,
        scale = 0.266,
    }

    model TowelBundle_BucketCarved
    {
        mesh = TowelBundle,
        texture = TowelBundleModel,
        scale = 0.336,
    }

    /* Thrown like a vanilla clay pot, but lower and wider, with a lid. Pottery 2. */
    item YoghurtPotUnfired
    {
        DisplayCategory = Material,
        ItemType = base:normal,
        Weight = 1.5,
        Icon = YoghurtPotUnfired,
        StaticModel = ClayJarUnfired,
        WorldStaticModel = ClayJarUnfired,
        Tooltip = Tooltip_TraditionalPreservationDairy_YoghurtPotUnfired,
    }

    item YoghurtPot
    {
        DisplayCategory = Material,
        ItemType = base:normal,
        Weight = 1.5,
        Icon = YoghurtPot,
        StaticModel = ClayJar,
        WorldStaticModel = ClayJar,
        Tooltip = Tooltip_TraditionalPreservationDairy_YoghurtPot,
    }

    /* The Yogurt Pot wrapped in a sheet: it carries scalded milk from the Hearth and holds its heat (about 15 C lost
       in an hour). Its milk's heat and scalded mark are in ModData tpdCarry, not in vanilla's item heat. */
    item WrappedYoghurtPot
    {
        DisplayCategory = Cooking,
        ItemType = base:normal,
        Weight = 2.5,
        Icon = WrappedYoghurtPot,
        StaticModel = TraditionalPreservationDairy.TowelBundle_ClayJar,
        WorldStaticModel = TraditionalPreservationDairy.TowelBundle_ClayJar,
        Tooltip = Tooltip_TraditionalPreservationDairy_WrappedYoghurtPot,
        component FluidContainer
        {
            ContainerName = WrappedYoghurtPot,
            RainFactor = 0.0,
            Capacity = 5.0,
            TransferRate = 5.0,
        }
    }

    /* Set yogurt in a glass jar (1 L) or a Clay Jar (2.5 L). Its food values are its portions, a vanilla Yogurt
       each per 250 ml, set in Lua when it is filled (none here, so a jar from before F4 is told by tpdFood in its
       ModData). The jar's type and litres are in ModData (tpdJar, tpdMl); its age is the yogurt's, carried over
       from the incubator. Eaten from, or spooned into cups from its right-click menu. */
    item JarOfYoghurt
    {
        DisplayCategory = Food,
        ItemType = base:food,
        FoodType = NoExplicit,
        DaysFresh = 10,
        DaysTotallyRotten = 15,
        Weight = 1.3,
        Icon = JarOfYoghurt,
        StaticModel = ClayJar,
        WorldStaticModel = ClayJar,
        Tooltip = Tooltip_TraditionalPreservationDairy_JarOfYoghurt,
    }

    /* The crock's grain-grown culture: one sets one batch. */
    item ActiveYoghurtStarter
    {
        DisplayCategory = Food,
        ItemType = base:food,
        CantEat = true,
        FoodType = NoExplicit,
        DaysFresh = 5,
        DaysTotallyRotten = 10,
        Weight = 0.2,
        Icon = ActiveYoghurtStarter,
        StaticModel = Yogurt,
        WorldStaticModel = Yogurt,
        Tooltip = Tooltip_TraditionalPreservationDairy_ActiveYoghurtStarter,
    }

    /* Every vanilla Yogurt leaves one when eaten or stirred into oatmeal (ReplaceOnUse, set in Lua). */
    item EmptyYoghurtContainer
    {
        DisplayCategory = Junk,
        ItemType = base:normal,
        Weight = 0.02,
        Icon = EmptyYoghurtContainer,
        StaticModel = Yogurt,
        WorldStaticModel = Yogurt,
        Tooltip = Tooltip_TraditionalPreservationDairy_EmptyYoghurtContainer,
    }
}
