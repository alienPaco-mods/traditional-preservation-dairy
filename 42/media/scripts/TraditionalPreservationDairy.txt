module TraditionalPreservationDairy
{
    imports
    {
        Base,
    }

    craftRecipe MakeYoghurtPot
    {
        timedAction = Craft_PotteryWheel,
        time = 120,
        Tags = PotteryWheel,
        SkillRequired = Pottery:2,
        xpAward = Pottery:20,
        category = Pottery,
        inputs
        {
            item 6 [Base.Clay],
            item 1 [*] mode:keep flags[ItemIsFluid;HandcraftOnly],
            -fluid 0.25 categories[Water] mode:mixture,
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPotUnfired,
        }
    }

    craftRecipe FireYoghurtPot
    {
        time = 20,
        timedAction = Making_With_Kiln,
        Tags = KilnSmall;KilnLarge,
        category = Pottery,
        inputs
        {
            item 1 [TraditionalPreservationDairy.YoghurtPotUnfired],
            item 1 tags[base:startfire],
            item 1 tags[base:log;base:charcoal],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPot,
        }
    }

    craftRecipe WrapYoghurtPot
    {
        timedAction = Making,
        time = 60,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.YoghurtPot],
            item 1 [Base.Sheet],
            item 1 [Base.Twine],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.WrappedYoghurtPot,
        }
    }

    /* Only when empty, so no milk is lost. */
    craftRecipe UnwrapYoghurtPot
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.WrappedYoghurtPot] mode:destroy flags[IsEmpty],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPot,
            item 1 Base.Sheet,
            item 1 Base.Twine,
        }
    }
}
