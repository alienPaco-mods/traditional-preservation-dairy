module TraditionalPreservationDairy
{
    imports
    {
        Base,
    }

    craftRecipe MakeYoghurtPot
    {
        timedAction = Craft_PotteryWheel,
        time = 120,
        Tags = PotteryWheel,
        SkillRequired = Pottery:2,
        xpAward = Pottery:20,
        category = Pottery,
        inputs
        {
            item 6 [Base.Clay],
            item 1 [*] mode:keep flags[ItemIsFluid;HandcraftOnly],
            -fluid 0.25 categories[Water] mode:mixture,
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPotUnfired,
        }
    }

    craftRecipe FireYoghurtPot
    {
        time = 20,
        timedAction = Making_With_Kiln,
        Tags = KilnSmall;KilnLarge,
        category = Pottery,
        inputs
        {
            item 1 [TraditionalPreservationDairy.YoghurtPotUnfired],
            item 1 tags[base:startfire],
            item 1 tags[base:log;base:charcoal],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPot,
        }
    }

    craftRecipe WrapYoghurtPot
    {
        timedAction = Making,
        time = 60,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.YoghurtPot],
            item 1 [Base.Sheet],
            item 1 [Base.Twine],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.WrappedYoghurtPot,
        }
    }

    /* Only when empty, so no milk is lost. */
    craftRecipe UnwrapYoghurtPot
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.WrappedYoghurtPot] mode:destroy flags[IsEmpty],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPot,
            item 1 Base.Sheet,
            item 1 Base.Twine,
        }
    }

    /* Any pot, jar or bucket, empty, wrapped the same way: it keeps its capacity and carries milk hot (M.wrap marks
       the item itself; M.WRAPPABLE lists the same items). Its wrap burns in a hot stove or fire. */
    craftRecipe WrapMilkCarrier
    {
        timedAction = Making,
        time = 60,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationDairy.canWrap,
        OnCreate = TraditionalPreservationDairy.onWrapped,
        inputs
        {
            item 1 [Base.Pot;Base.PotForged;Base.EmptyJar;Base.JarCrafted;Base.ClayJar;Base.Bucket;Base.BucketEmpty;Base.BucketCarved] mode:keep flags[IsEmpty],
            item 1 [Base.Sheet],
            item 1 [Base.Twine],
        }
        outputs
        {
        }
    }

    craftRecipe UnwrapMilkCarrier
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationDairy.canUnwrap,
        OnCreate = TraditionalPreservationDairy.onUnwrapped,
        inputs
        {
            item 1 [Base.Pot;Base.PotForged;Base.EmptyJar;Base.JarCrafted;Base.ClayJar;Base.Bucket;Base.BucketEmpty;Base.BucketCarved] mode:keep flags[IsEmpty],
        }
        outputs
        {
            item 1 Base.Sheet,
            item 1 Base.Twine,
        }
    }
}
