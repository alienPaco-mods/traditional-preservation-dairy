module TraditionalPreservationDairy
{
    imports
    {
        Base,
    }

    craftRecipe MakeYoghurtPot
    {
        timedAction = Craft_PotteryWheel,
        time = 120,
        Tags = PotteryWheel,
        SkillRequired = Pottery:2,
        xpAward = Pottery:20,
        category = Pottery,
        inputs
        {
            item 6 [Base.Clay],
            item 1 [*] mode:keep flags[ItemIsFluid;HandcraftOnly],
            -fluid 0.25 categories[Water] mode:mixture,
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPotUnfired,
        }
    }

    craftRecipe FireYoghurtPot
    {
        time = 20,
        timedAction = Making_With_Kiln,
        Tags = KilnSmall;KilnLarge,
        category = Pottery,
        inputs
        {
            item 1 [TraditionalPreservationDairy.YoghurtPotUnfired],
            item 1 tags[base:startfire],
            item 1 tags[base:log;base:charcoal],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPot,
        }
    }

    craftRecipe WrapYoghurtPot
    {
        timedAction = Making,
        time = 60,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.YoghurtPot],
            item 1 [Base.Sheet],
            item 1 [Base.Twine],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.WrappedYoghurtPot,
        }
    }

    /* Only when empty, so no milk is lost. */
    craftRecipe UnwrapYoghurtPot
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.WrappedYoghurtPot] mode:destroy flags[IsEmpty],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPot,
            item 1 Base.Sheet,
            item 1 Base.Twine,
        }
    }

    /* Any pot, jar or bucket, empty, wrapped the same way: it keeps its capacity and carries milk hot (M.wrap marks
       the item itself; M.WRAPPABLE lists the same items). Its wrap burns in a hot stove or fire. */
    craftRecipe WrapMilkCarrier
    {
        timedAction = Making,
        time = 60,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationDairy.canWrap,
        OnCreate = TraditionalPreservationDairy.onWrapped,
        inputs
        {
            item 1 [Base.Pot;Base.PotForged;Base.EmptyJar;Base.JarCrafted;Base.ClayJar;Base.Bucket;Base.BucketEmpty;Base.BucketCarved] mode:keep flags[IsEmpty],
            item 1 [Base.Sheet],
            item 1 [Base.Twine],
        }
        outputs
        {
        }
    }

    craftRecipe UnwrapMilkCarrier
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationDairy.canUnwrap,
        OnCreate = TraditionalPreservationDairy.onUnwrapped,
        inputs
        {
            item 1 [Base.Pot;Base.PotForged;Base.EmptyJar;Base.JarCrafted;Base.ClayJar;Base.Bucket;Base.BucketEmpty;Base.BucketCarved] mode:keep flags[IsEmpty],
        }
        outputs
        {
            item 1 Base.Sheet,
            item 1 Base.Twine,
        }
    }

    /* Butter that keeps (TraditionalPreservationDairyButter.lua). Salting takes a salt (anything tagged salt, as Fish
       & Meat's cures do) worked into a butter: it keeps three times as long. Clarifying takes no station: a pot,
       one or two butter, then any stove or fire, and the clear fat poured off its milk solids keeps as Lard does.
       The Stone Hearth's cauldron clarifies many at once (Put Butter in the Cauldron, then Strain Clarified Butter).
       Every step but scraping a burnt pot takes Cooking 2. */
    craftRecipe SaltButter
    {
        timedAction = MixingBowl,
        time = 50,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        SkillRequired = Cooking:2,
        OnCreate = TraditionalPreservationDairy.onSaltButter,
        inputs
        {
            item 1 [Base.Butter] flags[ItemCount] mode:destroy,
            item 1 tags[base:salt],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.SaltedButter,
        }
    }

    craftRecipe PutButterInPot
    {
        timedAction = MixingBowl,
        time = 40,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        SkillRequired = Cooking:2,
        OnCreate = TraditionalPreservationDairy.onPutButterInPot,
        inputs
        {
            item 1 [Base.Pot;Base.PotForged] mode:destroy flags[InheritCondition;IsEmpty] mappers[potType],
            item 1 [Base.Butter;TraditionalPreservationDairy.SaltedButter] flags[InheritFoodAge;ItemCount] mode:destroy,
        }
        outputs
        {
            item 1 mapper:potType,
        }
        itemMapper potType
        {
            TraditionalPreservationDairy.PotOfButter = Base.Pot,
            TraditionalPreservationDairy.PotForgedButter = Base.PotForged,
        }
    }

    craftRecipe AddButterToPot
    {
        timedAction = MixingBowl,
        time = 30,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        SkillRequired = Cooking:2,
        OnTest = TraditionalPreservationDairy.canAddButter,
        OnCreate = TraditionalPreservationDairy.onAddButter,
        inputs
        {
            item 1 [TraditionalPreservationDairy.PotOfButter;TraditionalPreservationDairy.PotForgedButter] mode:destroy flags[InheritCondition] mappers[sameItem],
            item 1 [Base.Butter;TraditionalPreservationDairy.SaltedButter] flags[ItemCount] mode:destroy,
        }
        outputs
        {
            item 1 mapper:sameItem,
        }
        itemMapper sameItem
        {
            TraditionalPreservationDairy.PotOfButter = TraditionalPreservationDairy.PotOfButter,
            TraditionalPreservationDairy.PotForgedButter = TraditionalPreservationDairy.PotForgedButter,
        }
    }

    /* A cooked pot: the clear fat poured off its milk solids through a cloth or a Strainer (kept), a Clarified Butter
       for each butter that went in (onPourOffClarified adds the second), and the pot back. IsCookedFoodItem lets burnt food through, so canPourOffClarified refuses a
       burnt pot. */
    craftRecipe PourOffClarifiedButter
    {
        timedAction = MixingBowl,
        time = 60,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        SkillRequired = Cooking:2,
        xpAward = Cooking:5,
        OnTest = TraditionalPreservationDairy.canPourOffClarified,
        OnCreate = TraditionalPreservationDairy.onPourOffClarified,
        inputs
        {
            item 1 [TraditionalPreservationDairy.PotOfButter;TraditionalPreservationDairy.PotForgedButter] mappers[potType] flags[InheritCondition;IsCookedFoodItem;ItemCount] mode:destroy,
            item 1 [Base.CheeseCloth;Base.DishCloth;Base.RippedSheets;Base.Sheet;Base.Strainer] mode:keep,
        }
        outputs
        {
            item 1 mapper:potType,
            item 1 TraditionalPreservationDairy.ClarifiedButter,
        }
        itemMapper potType
        {
            Base.Pot = TraditionalPreservationDairy.PotOfButter,
            Base.PotForged = TraditionalPreservationDairy.PotForgedButter,
        }
    }

    /* Clarified butter melted in the Stone Hearth's cauldron, in whatever it was ladled into: a quarter litre strains
       through a cloth or a Strainer (kept, as the container is) to a Clarified Butter, as a butter in a pot does. */
    craftRecipe StrainClarifiedButter
    {
        timedAction = MixingBowl,
        time = 40,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        SkillRequired = Cooking:2,
        xpAward = Cooking:3,
        inputs
        {
            item 1 [Base.CheeseCloth;Base.DishCloth;Base.RippedSheets;Base.Sheet;Base.Strainer] mode:keep,
            item 1 [*],
            -fluid 0.25 [MeltedClarifiedButter],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.ClarifiedButter,
        }
    }

    /* The only way to get the pot back from burnt butter (canScrapeButter), rotten or not. */
    craftRecipe ScrapeOutBurntButter
    {
        timedAction = MixingBowl,
        time = 80,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationDairy.canScrapeButter,
        inputs
        {
            item 1 [TraditionalPreservationDairy.PotOfButter;TraditionalPreservationDairy.PotForgedButter] mappers[potType] flags[InheritCondition;AllowRottenItem;ItemCount] mode:destroy,
        }
        outputs
        {
            item 1 mapper:potType,
        }
        itemMapper potType
        {
            Base.Pot = TraditionalPreservationDairy.PotOfButter,
            Base.PotForged = TraditionalPreservationDairy.PotForgedButter,
        }
    }
}
