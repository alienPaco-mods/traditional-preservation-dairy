module TraditionalPreservationDairy
{
    imports
    {
        Base,
    }

    craftRecipe MakeYoghurtPot
    {
        timedAction = Craft_PotteryWheel,
        time = 120,
        Tags = PotteryWheel,
        SkillRequired = Pottery:2,
        xpAward = Pottery:20,
        category = Pottery,
        inputs
        {
            item 6 [Base.Clay],
            item 1 [*] mode:keep flags[ItemIsFluid;HandcraftOnly],
            -fluid 0.25 categories[Water] mode:mixture,
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPotUnfired,
        }
    }

    craftRecipe FireYoghurtPot
    {
        time = 20,
        timedAction = Making_With_Kiln,
        Tags = KilnSmall;KilnLarge,
        category = Pottery,
        inputs
        {
            item 1 [TraditionalPreservationDairy.YoghurtPotUnfired],
            item 1 tags[base:startfire],
            item 1 tags[base:log;base:charcoal],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPot,
        }
    }

    craftRecipe WrapYoghurtPot
    {
        timedAction = Making,
        time = 60,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.YoghurtPot],
            item 1 [Base.Sheet],
            item 1 [Base.Twine],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.WrappedYoghurtPot,
        }
    }

    /* Only when empty, so no milk is lost. */
    craftRecipe UnwrapYoghurtPot
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.WrappedYoghurtPot] mode:destroy flags[IsEmpty],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.YoghurtPot,
            item 1 Base.Sheet,
            item 1 Base.Twine,
        }
    }

    /* Any pot, jar or bucket, empty, wrapped the same way: it keeps its capacity and carries milk hot (M.wrap marks
       the item itself; M.WRAPPABLE lists the same items). Its wrap burns in a hot stove or fire. */
    craftRecipe WrapMilkCarrier
    {
        timedAction = Making,
        time = 60,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationDairy.canWrap,
        OnCreate = TraditionalPreservationDairy.onWrapped,
        inputs
        {
            item 1 [Base.Pot;Base.PotForged;Base.EmptyJar;Base.JarCrafted;Base.ClayJar;Base.ClayJarGlazed;Base.Bucket;Base.BucketEmpty;Base.BucketCarved] mode:keep flags[IsEmpty],
            item 1 [Base.Sheet],
            item 1 [Base.Twine],
        }
        outputs
        {
        }
    }

    craftRecipe UnwrapMilkCarrier
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationDairy.canUnwrap,
        OnCreate = TraditionalPreservationDairy.onUnwrapped,
        inputs
        {
            item 1 [Base.Pot;Base.PotForged;Base.EmptyJar;Base.JarCrafted;Base.ClayJar;Base.ClayJarGlazed;Base.Bucket;Base.BucketEmpty;Base.BucketCarved] mode:keep flags[IsEmpty],
        }
        outputs
        {
            item 1 Base.Sheet,
            item 1 Base.Twine,
        }
    }

    /* Butter that keeps (TraditionalPreservationDairyButter.lua). Salting takes a salt (anything tagged salt, as Fish
       & Meat's cures do) worked into a butter: it keeps three times as long. Clarifying takes no station: Put Butter
       in a Pot (a menu, not a recipe: a pot or saucepan, or the Stone Hearth's cauldron for many at once) melts it to
       a fluid, steamed an hour on any stove or fire it is clarified, then Strain Clarified Butter. Every step but
       scraping a burnt pot takes Cooking 2. */
    craftRecipe SaltButter
    {
        timedAction = MixingBowl,
        time = 50,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        SkillRequired = Cooking:2,
        OnCreate = TraditionalPreservationDairy.onSaltButter,
        inputs
        {
            item 1 [Base.Butter] flags[ItemCount] mode:destroy,
            item 1 tags[base:salt],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.SaltedButter,
        }
    }

    /* Clarified butter melted in a pot, a saucepan or the Stone Hearth's cauldron, in whatever holds it: a quarter
       litre strains through a cloth or a Strainer (kept, as the container is) to a Clarified Butter. */
    craftRecipe StrainClarifiedButter
    {
        timedAction = MixingBowl,
        time = 40,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        SkillRequired = Cooking:2,
        xpAward = Cooking:3,
        inputs
        {
            item 1 [Base.CheeseCloth;Base.DishCloth;Base.RippedSheets;Base.Sheet;Base.Strainer] mode:keep,
            item 1 [*],
            -fluid 0.25 [MeltedClarifiedButter],
        }
        outputs
        {
            item 1 TraditionalPreservationDairy.ClarifiedButter,
        }
    }

    /* The only way to get the vessel back from burnt butter, its condition kept. */
    craftRecipe ScrapeOutBurntButter
    {
        timedAction = MixingBowl,
        time = 80,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.PotOfBurntButter;TraditionalPreservationDairy.PotForgedBurntButter;TraditionalPreservationDairy.SaucepanBurntButter;TraditionalPreservationDairy.SaucepanCopperBurntButter] mappers[vesselType] flags[InheritCondition;AllowRottenItem;ItemCount] mode:destroy,
        }
        outputs
        {
            item 1 mapper:vesselType,
        }
        itemMapper vesselType
        {
            Base.Pot = TraditionalPreservationDairy.PotOfBurntButter,
            Base.PotForged = TraditionalPreservationDairy.PotForgedBurntButter,
            Base.Saucepan = TraditionalPreservationDairy.SaucepanBurntButter,
            Base.SaucepanCopper = TraditionalPreservationDairy.SaucepanCopperBurntButter,
        }
    }

    /* Opened as vanilla opens a seed packet. The packet isn't kept: it is a sachet torn open. */
    craftRecipe OpenKefirGrainsPacket
    {
        timedAction = PutSeedsInPacket,
        time = 20,
        Tags = InHandCraft;CanBeDoneFromFloor;CanBeDoneInDark;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [TraditionalPreservationDairy.KefirGrainsPacket] flags[ItemCount],
        }
        outputs
        {
            item 10 TraditionalPreservationDairy.DriedKefirGrains,
        }
    }
}
